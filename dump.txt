    if (!token) { setGistResult('请先填入 GitHub Token', false); return; }
    if (records.length === 0 && categories.length === 0 && themes.length === 0) {
      setGistResult('当前没有数据可备份', false);
      return;
    }
    const repo = (settings.gistId || '').trim() || BACKUP_REPO_DEFAULT;
    setGistResult('正在准备备份…', null);

    buildBackup().then(backup => {
      const content = JSON.stringify(backup, null, 2);
      let encoded;
      try {
        encoded = btoa(unescape(encodeURIComponent(content)));
      } catch (e) {
        setGistResult('备份内容编码失败', false);
        return;
      }
      setGistResult('正在上传…', null);

      const apiPath = '/repos/' + BACKUP_REPO_OWNER + '/' + repo + '/contents/' + BACKUP_FILE_PATH;

      // 先看看文件在不在（在的话要带 sha 才能更新）

        .then((file) => ghRequest('PUT', apiPath, token, {
          message: '备份 ' + new Date().toISOString(),
          content: encoded,
          sha: file.sha
        }))
        .catch((err) => {
          // 文件不存在（404）→ 直接创建
          if (err && err.status === 404) {
            return ghRequest('PUT', apiPath, token, {
              message: '备份 ' + new Date().toISOString(),
              content: encoded
            });
          }
          throw err;
        })
        .then(() => {
          settings.gistId = repo;
          saveSettings();
          updateGistHint();
          setGistResult('上传成功：' + records.length + ' 条记录、' + categories.length + ' 个分类' + (themes.length ? '、' + themes.length + ' 个主题' : '') + '\n仓库：' + BACKUP_REPO_OWNER + '/' + repo, true);
        });
    }).catch((e) => {
      setGistResult('上传失败：' + e.message, false);

  function handleGistDownload() {
    gistSaveSettings();
    const token = (settings.gistToken || '').trim();
    if (!token) { setGistResult('请先填入 GitHub Token', false); return; }
    const repo = (settings.gistId || '').trim() || BACKUP_REPO_DEFAULT;
    const mode = settings.restoreMode || 'merge';
    setGistResult('正在下载…（' + (mode === 'overwrite' ? '全部覆盖' : '只新增') + '）', null);

    const apiPath = '/repos/' + BACKUP_REPO_OWNER + '/' + repo + '/contents/' + BACKUP_FILE_PATH;

    ghRequest('GET', apiPath, token)
      .then(async (file) => {
        if (!file || !file.content) throw new Error('仓库里没找到备份文件');
        const b64 = String(file.content).replace(/\n/g, '');
        let content;
        try {
          content = decodeURIComponent(escape(atob(b64)));
        } catch (e) {
          throw new Error('备份内容解码失败');
        }
        let raw;
        try { raw = JSON.parse(content); } catch (err) { throw new Error('备份内容不是有效 JSON'); }
        const bk = extractBackup(raw);
        if (!bk) throw new Error('备份格式不对');

        const res = await applyBackup(bk, mode);
        renderFilterRow();
        renderCategories();
        renderThemes();
        render();
        setGistResult('恢复完成：' + summarizeBackup(res).replace(/^✓ /, ''), true);
      })
      .catch((e) => {
        setGistResult('恢复失败：' + e.message, false);
      });
  }
